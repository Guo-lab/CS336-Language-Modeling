# Train a language model on the TinyStories dataset using CUDA.  
```
CUDA_VISIBLE_DEVICES=0 uv run scripts/train_lm.py \
  --train-data artifacts/lm_data/tinystories_train_10k.npy --valid-data artifacts/lm_data/tinystories_valid_10k.npy \
  --tokenizer artifacts/tokenizer_experiments/training/tokenizers_chunked_mp8/tinystories_train_10k \
  --run-name tinystories_lr_2em3_cuda_$(date +%Y%m%d) --out-dir artifacts/lm_experiments/cuda_baseline --device cuda --seed 0 --vocab-size 10000 \
  --context-length 256 --batch-size 32 --num-layers 4 --d-model 512 --num-heads 16 --d-ff 1344 \
  --max-iters 5000 --warmup-iters 500 --cosine-cycle-iters 5000 --max-lr 2e-3 --min-lr 2e-4 \
  --weight-decay 0.1 --beta1 0.9 --beta2 0.95 --eps 1e-8 --grad-clip 1.0 \
  --log-every 100 --eval-every 1000 --eval-iters 20 --save-every 2500 --sample-every 0 \
  --wandb-project cs336-a1 --wandb-mode offline
```

# Train a language model on the OWT dataset using CUDA.
```
CUDA_VISIBLE_DEVICES=0 uv run scripts/train_lm.py \
  --train-data artifacts/lm_data/owt_train_32k.npy --valid-data artifacts/lm_data/owt_valid_32k.npy \
  --tokenizer artifacts/tokenizer_experiments/training/tokenizers_chunked_mp8/owt_train_32k \
  --run-name owt_large_l6_d640_h10_bs16_cuda_$(date +%Y%m%d) --out-dir artifacts/lm_experiments/cuda_baseline \
  --device cuda --seed 0 --vocab-size 32000 \
  --context-length 256 --batch-size 16 --num-layers 6 --d-model 640 --num-heads 10 --d-ff 1728 \
  --max-iters 10000 --warmup-iters 1000 --cosine-cycle-iters 10000 --max-lr 1e-3 --min-lr 1e-4 \
  --weight-decay 0.1 --beta1 0.9 --beta2 0.95 --eps 1e-8 --grad-clip 1.0 \
  --log-every 200 --eval-every 2000 --eval-iters 200 --save-every 5000 \
  --sample-every 2500 --sample-prompt "The" --sample-max-new-tokens 120 --sample-temperature 0.8 --sample-top-p 0.9 \
  --wandb-project cs336-a1 --wandb-mode offline
```
